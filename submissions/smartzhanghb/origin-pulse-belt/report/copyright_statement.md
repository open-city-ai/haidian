f_body = ImageFont.truetype(FP, 34)
f_body_b = ImageFont.truetype(FP, 36)
f_foot = ImageFont.truetype(FP, 24)


def text(d, xy, s, font, fill):
    d.text(xy, s, font=font, fill=fill)


def autocrop(path, pad=50):
    """裁掉 PNG 四周接近背景色的空白，返回新文件路径"""
    im = Image.open(path).convert('RGB')
    gray = im.convert('L')
    bbox = gray.point(lambda v: 0 if v > 235 else 255).getbbox()
    if not bbox:
        return path
    x0, y0, x1, y1 = bbox
    x0, y0 = max(0, x0-pad), max(0, y0-pad)
    x1, y1 = min(im.width, x1+pad), min(im.height, y1+pad)
    out = f'/tmp/crop_{path.split("/")[-1]}'
    im.crop((x0, y0, x1, y1)).save(out)
    return out


def paste_fig(canvas, path, box):
    """把 PNG 按等比缩放放进 box（左上对齐），返回实际占用的 (w,h)"""
    fig = Image.open(path)
    bw, bh = box[2] - box[0], box[3] - box[1]
    r = min(bw / fig.width, bh / fig.height)
    w, h = int(fig.width * r), int(fig.height * r)
    fig = fig.resize((w, h), Image.LANCZOS)
    canvas.paste(fig, (box[0] + (bw - w) // 2, box[1] + (bh - h) // 2))
    return w, h


def new_page(title, subtitle):
    im = Image.new('RGB', (W, H), BG)

    text(d, (M, M - 30), title, f_h1, ACC)

    text(d, (W - M - d.textlength(foot_r, font=f_foot), H - M + 10), foot_r, f_foot, GREY)
    return im, d

    """右侧文字卡：标题 + 若干行"""
    text(d, (x, y), title, f_body_b, ACC)
    d.line([(x, y + 56), (x + w, y + 56)], fill=LINE, width=2)
    yy = y + 100
    for ln in lines:
        text(d, (x, yy), ln, f_body, INK)
        yy += gap
    return yy


pages = []

yy = card(d, tx, yy + 80, tw, '协同回路', [
    '高校策源 → 全栈攻坚 → 场景变现',

yy = card(d, tx, yy + 80, tw, 'Logo 方向', [
    '京张「人」字线抽象三股轨道线，',

])
# 底部设计判断条
jy = top + avail_h + 40
d.line([(M, jy), (W - M, jy)], fill=LINE, width=2)
text(d, (M, jy + 24), '设计判断：走廊不是剩余空间，而是组织一切的骨架——绿地、慢行、场景、展示全部沿轴展开；三核按「全栈攻坚/近校转化/原生业态」分工互补；两片混合街坊保证职住商服均衡。本板所有空间关系可从 geometry/*.geojson 复算。', f_body, INK)
pages.append(im)

# ---------- P2 用地布局 ----------
im, d = new_page('用地布局与城市更新板', '56 街坊无缝覆盖 · 概念配比 · 拆改留分类逻辑')
top = M + 220

paste_fig(im, f'{FIG}/land-use-structure.png', (M, top, M + fig_w, top + avail_h))
tx = M + fig_w + 90
